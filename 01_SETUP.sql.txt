-- CREATE WAREHOUSE , DATABASE AND SCHEMA
CREATE OR REPLACE WAREHOUSE MOCK_WH
WITH 
WAREHOUSE_SIZE = 'X-SMALL'
AUTO_SUSPEND = 60
AUTO_RESUME = TRUE;

SHOW WAREHOUSES;

CREATE DATABASE IF NOT EXISTS MOCK_DB;

SHOW DATABASES;

CREATE SCHEMA RAW;
CREATE SCHEMA CORE;
CREATE SCHEMA MART;

SHOW SCHEMAS;
